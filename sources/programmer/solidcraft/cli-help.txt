usage:
  solidcraft-cli run <script.json|design.sc|part.step> [--in design.sc|part.step] [--out FILE]... [--save FILE] [--quiet] [--results] [--timings] [--profile] [--profile-out trace.json] [--json-out FILE] [--allow-errors] [--validation each_step|each_feature|off]
                     [--param-check off|warn|strict] [--quality standard|fine|finest] [--script-relative]
  solidcraft-cli run --in <design.sc|part.step> [--out FILE]...
      (a timeline feature that fails fails the run and nothing is written; --allow-errors writes anyway)
      (--quality: how closely an STL, OBJ or 3MF --out follows curved faces, as file.export's quality: a chord of
       2e-4, 5e-5 or 1e-5 of each body's size, never under 1 µm, default standard)
      (--validation: when SMLib's full check runs on the features computed (diagnostics.validation), default each_feature)
      (--param-check checks each step against its command's contract: warn reports on stderr, strict stops the run)
      (--script-relative runs command file paths from the positional script's folder; command-line file paths
       stay relative to the caller's folder, which remains the default for scripts without this option)
      (--timings prints {timings: [{index, command, ms}], total_ms} for a script, with --results its results too;
       --profile adds each command's profile: {boolean_ms, tessellate_ms, other_ms}, the kernel's share)
      (--profile also records where the time goes, SMLib's calls, kernel operations, features, commands and import stages,
       and prints the table to stderr: count, total, self time and longest per span; --profile-out FILE writes the spans
       as a Chrome trace for chrome://tracing or ui.perfetto.dev; SOLIDCRAFT_PROFILE=1 does the same for any command)
  solidcraft-cli eval <script.json|design.sc|part.step> [--in design.sc|part.step] [--json-out FILE] [--profile] [--profile-out trace.json] [--validation each_step|each_feature|off]     (alias: inspect)
  solidcraft-cli eval --in <script.json|design.sc|part.step> [--json-out FILE]
      (with both inputs, runs the positional script on --in before measuring; does not automatically save the modified design)
      (--json-out FILE writes the JSON that run (with --results, else the measurement) and eval print to FILE instead of stdout,
       as plain UTF-8: a shell's > can re-encode it, e.g. Windows PowerShell adds a byte-order mark)
      (--profile, --profile-out and --validation as for run)
  solidcraft-cli script <design.sc|script.json> [--out script.json]
      (the design's history as a command script that eval and run replay)
  solidcraft-cli bench-open <design.sc | part.step> [--parse-only] [--save OUT.sc] [--display LEVEL [--edit NAME EXPR]] [--ops script.json] [--profile] [--profile-out trace.json]
      (--profile and --profile-out as for run)
      (--ops: after opening, run each command of the script and undo it, timing the command, the
       evaluations it caused and the view it leaves: placing and meshing what changed)
  solidcraft-cli bench-agent [--scale F] [--save design.sc] [--script-out script.json]
  solidcraft-cli bench-assistant --connect 127.0.0.1:PORT [--token T] --base-url URL --model NAME [--tasks 1,2,5] [--repeat N] [--timeout-s 300] [--json out.json] [--markdown out.md]
  solidcraft-cli snapshot <script|design> [--in design.sc|part.step] --out shot.png [--width W] [--height H] [--view NAME | --camera JSON] [--no-grid] [--no-axes]
      [--body NAME:opacity=0.3,color=#rrggbb]... [--tool NAME]... [--section z=27] [--quality coarse|normal|fine|ultra]
  solidcraft-cli snapshot --in <design.sc|part.step> --out shot.png ...
      (as with run, --in starts from a design and the script runs on top of it; an option snapshot doesn't take is an error;
       --view: iso, front, back, top, bottom, left, right, or the name of a view saved in the design with view.save;
       --camera: {target: [x,y,z], yaw, pitch, distance, fov?} in mm and radians, as the MCP screenshot takes it (docs/mcp.md);
       --no-grid and --no-axes leave the ground grid (on Z 0) and the origin's axes out of the picture;
       --body: how one body is drawn, as appearance.assign takes it: opacity 0-1 (below 1 is see-through, so what is inside
         it shows) and color #rrggbb, either or both, once per body;
       --tool: draw this body solid in the tool colour and every other body see-through, so a cut's or a join's tool body
         is visible inside the material it meets; a later --body overrides it;
       --section: cut the model as inspect.section does, so the inside shows: z=27 (the plane through that world
         coordinate, keeping the smaller side), add ,flip to keep the other side, or XY, XZ, YZ, or its own JSON;
       --quality: the display quality the bodies are meshed at, as the app's setting (default normal))
  solidcraft-cli render <script|design> --out image.png [--setup NAME] [--width W] [--height H] [--transparent] [--overwrite] [--timeout SECONDS]
      [--progressive] [--samples N] [--max-depth N] [--seed N] [--exposure EV] [--lighting FILE.json] [--environment FILE.json] [--tone-map reinhard|aces_fitted] [--legacy-memory-admission]
      --legacy-memory-admission skips available-RAM checks; fixed shared/host budgets and allocation limits still apply.
  solidcraft-cli exec <command> [json-params]
  solidcraft-cli commands
  solidcraft-cli recipe <recipe.json>
  solidcraft-cli oracle <case-dir>... [--json]
  solidcraft-cli kernel-cases <case-dir>... [--json] [--dumps=DIR] [--export=DIR] [--validation=each_step|each_feature|off]
      (the kernel regression rig, tests/kernel-cases, on SMLib; --dumps: each case's
       failing SMLib calls leave their inputs for replay in DIR/<area>-<case>; --export: each case's
       kernel operations, input records and expected values for solidcore's rig, in DIR/<area>/<case>.json; --validation: when SMLib's
       full check runs (diagnostics.validation), default each_feature)
  solidcraft-cli fuzz [--from N] [--count N] [--jobs J] [--timeout S] [--steps N] [--minimise] [--out DIR] [--report FILE]
  solidcraft-cli mcp [--headless] [--in design.sc|script.json] [--connect 127.0.0.1:PORT [--token T]] [--allow-path DIR]... [--allow-any-path] [--require-allow-path]
                     [--param-check off|warn|strict] [--render-gpu] [--compact]
      (--compact, SOLIDCRAFT_MCP_COMPACT=1 or the app's Preferences › AI (with --connect): compact tool output for
       agents: inspect_design summarises sketches and takes `sketches` by name and a `bodies` filter, and tool JSON
       is not pretty-printed; off by default)
  solidcraft-cli mcp setup [--dry-run] [--client opencode|claude-desktop|claude-code|cursor] [--live]
      (adds a `solidcraft` server, with this program's path, to each installed AI client; --live adds `solidcraft-live` for the running app; alias: mcp install)
  solidcraft-cli crashes [list [--json] | show <id> | replay <id> | dir | clear [<id>]]
  solidcraft-cli licences
  solidcraft-cli gpu [--policy auto|hardware|software|cpu]
      (the GPU adapters a headless render can open, best first, the one the policy picks and the tier it
       lands on: gpu, gpu-software, or the CPU rasterizer (auto with no adapter, or cpu); the policy defaults
       to SOLIDCRAFT_GPU, else auto; exits with an error when hardware or software finds no such adapter)
  solidcraft-cli icon --out icon.png [--size 256] | --out icon.ico
  --debug, with any command: what SolidCraft logs (why a body has no display plan, why a save left its
      model cache out), on stderr; RUST_LOG=error|warn|info|debug|trace sets one level for every crate
  --max-file-size SIZE, with any command: read STEP and import files and design files up to SIZE for this run
      (for example 2GB or 1500MB; at most 16GB; the defaults are 512 MB and 256 MB, and a smaller SIZE keeps them)
  --plugin-limits FILE.json, with local commands: user ceilings {max_fuel, max_memory_mib}; defaults 100000000000 / 128, invocation defaults remain 10000000 / 64 MiB. With mcp use --headless; connected desktops keep their own preferences.
